  function submitEntry(entry, cb) {
    try { sessionStorage.setItem(PENDING_KEY, JSON.stringify(entry)); } catch (e) {}

    var idx = state.responses.findIndex(function (r) { return r.id === entry.id; });
    var nextResponses = state.responses.slice();
    if (idx >= 0) nextResponses[idx] = entry; else nextResponses.push(entry);
    var nextState = { responses: nextResponses };

    if (!window.claude || !window.claude.use) {
      cb(false, 'L’enregistrement automatique n’est pas disponible ici. Merci de nous contacter directement pour confirmer.');
      return;
    }

    window.claude.use('artifact').then(function (artifact) {
      if (!artifact) {
        cb(false, 'L’enregistrement automatique n’est pas disponible ici. Merci de nous contacter directement pour confirmer.');
        return;
      }
      artifact.publish(buildFullDocument(nextState)).then(function () {
        try { sessionStorage.removeItem(PENDING_KEY); } catch (e) {}
        cb(true);
      }).catch(function (err) {
        var code = err && err.code;
        if (code === 'conflict') {
          cb(false, 'Une autre réponse vient d’être enregistrée, la page se met à jour. Merci de renvoyer votre réponse dans un instant.');
        } else if (code === 'not_writer' || code === 'not_granted' || code === 'consent_required') {
          cb(false, 'Cette page est en lecture seule pour vous : merci de confirmer votre présence directement auprès d’Axelle & Antonio.');
        } else {
          cb(false, 'Une erreur est survenue, merci de réessayer.');
        }
      });
    }).catch(function () {
      cb(false, 'Une erreur est survenue, merci de réessayer.');
    });
  }

  function renderConfirmation(slot, entry) {
    slot.innerHTML = '';
    var card = document.createElement('div');
    card.className = 'confirm-card';
    var h3 = document.createElement('h3');
    h3.textContent = entry.attending ? 'Merci, ' + entry.name + ' \u{1F90D}' : 'Merci pour votre réponse';
    var p = document.createElement('p');
    p.textContent = entry.attending
      ? 'Votre présence est bien enregistrée, avec ' + entry.guests.length + ' personne(s). À très bientôt !'
      : 'Nous sommes désolés que vous ne puissiez pas être là, merci de nous l’avoir fait savoir.';
    var editBtn = document.createElement('button');
    editBtn.type = 'button';
    editBtn.className = 'edit-link';
    editBtn.textContent = 'Modifier ma réponse';
    editBtn.addEventListener('click', function () { renderForm(slot, entry); });
    card.appendChild(h3);
    card.appendChild(p);
    card.appendChild(editBtn);
    slot.appendChild(card);
  }
